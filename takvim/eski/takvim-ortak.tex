% ---------- Ay ızgarası ----------
% \AyIzgara[masa]{ay}{hücre genişliği}{hücre yüksekliği}{yazı komutu}
% Seçeneksiz: gün numarası ortada, çerçevesiz (poster).
% [masa]: çerçeveli hücre, numara sol üstte, tatil adı hücrenin içinde.
\newcount\tkJ \newcount\tkIlk \newcount\tkN \newcount\tkD
\newcount\tkPos \newcount\tkSatir \newcount\tkSutun
\newif\iftkMasa
\newcommand{\AyIzgara}[5][poster]{%
  \def\tkMod{#1}\def\tkMasaAd{masa}%
  \ifx\tkMod\tkMasaAd\tkMasatrue\else\tkMasafalse\fi
  \pgfcalendardatetojulian{2027-#2-01}{\tkJ}%
  \pgfcalendarjuliantoweekday{\tkJ}{\tkIlk}% 0 = pazartesi
  \pgfcalendardatetojulian{2027-#2-last}{\tkN}%
  \advance\tkN by -\tkJ \advance\tkN by 1
  \begin{tikzpicture}[x=#3,y=#4,inner sep=0pt,outer sep=0pt]
    \path[use as bounding box] (0,1) rectangle (7,-6);
    \foreach \gun [count=\i from 0] in {Pzt,Sal,Çar,Per,Cum,Cmt,Paz}{%
      \ifnum\i>4
        \node[text=kizil] at (\i+0.5,0.45) {#5\Dar\bfseries\gun};
      \else
        \node[text=yumusak] at (\i+0.5,0.45) {#5\Dar\bfseries\gun};
      \fi}
    \draw[lacivert,line width=0.8pt] (0,0)--(7,0);
    \tkD=1
    \loop
      \tkPos=\tkIlk \advance\tkPos by \tkD \advance\tkPos by -1
      \tkSatir=\tkPos \divide\tkSatir by 7
      \tkSutun=\tkSatir \multiply\tkSutun by -7 \advance\tkSutun by \tkPos
      \edef\tkRenk{\ifnum\tkSutun>4 kizil\else lacivert\fi}%
      \def\tkAd{}\def\tkTur{0}%
      \ifcsname tatil@#2-\the\tkD\endcsname
        \edef\tkAd{\csname tatil@#2-\the\tkD\endcsname}\def\tkTur{1}\fi
      \ifcsname yarim@#2-\the\tkD\endcsname
        \edef\tkAd{\csname yarim@#2-\the\tkD\endcsname}\def\tkTur{2}\fi
      \iftkMasa
        \ifnum\tkTur=1 \fill[kizil!14] (\the\tkSutun,-\the\tkSatir) rectangle ++(1,-1);\fi
        \ifnum\tkTur=2 \fill[kizil!14] (\the\tkSutun,-\the\tkSatir)--++(1,0)--++(0,-1)--cycle;\fi
        \draw[izgara,line width=0.5pt] (\the\tkSutun,-\the\tkSatir) rectangle ++(1,-1);
        \ifnum\tkTur>0 \def\tkRenk{kizil}\fi
        \node[anchor=north west,text=\tkRenk] at (\the\tkSutun+0.09,-\the\tkSatir-0.1)
          {#5\Dar\bfseries\the\tkD};
        \ifnum\tkTur>0
          \node[anchor=south west,text=kizil] at (\the\tkSutun+0.09,-\the\tkSatir-0.9)
            {\fontsize{6.2}{7}\selectfont\Dar\tkAd};
        \fi
      \else
        \ifnum\tkTur=1
          \fill[kizil] (\the\tkSutun+0.5,-\the\tkSatir-0.5) circle[x radius=0.36,y radius=0.42];
          \def\tkRenk{white}\fi
        \ifnum\tkTur=2
          \draw[kizil,line width=0.7pt] (\the\tkSutun+0.5,-\the\tkSatir-0.5) circle[x radius=0.36,y radius=0.42];
          \def\tkRenk{kizil}\fi
        \node[text=\tkRenk] at (\the\tkSutun+0.5,-\the\tkSatir-0.5) {#5\the\tkD};
      \fi
    \ifnum\tkD<\tkN \advance\tkD by 1
    \repeat
  \end{tikzpicture}}
